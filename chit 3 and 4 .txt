
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Input

# 1. Load dataset
train_data = pd.read_csv("dataset/sign_mnist_train.csv")
test_data = pd.read_csv("dataset/sign_mnist_test.csv")

# 2. Separate features and labels
X_train = train_data.drop('label', axis=1)
y_train = train_data['label']

X_test = test_data.drop('label', axis=1)
y_test = test_data['label']

# 3. Normalize pixel values
X_train = X_train / 255.0
X_test = X_test / 255.0

# 4. Convert into NumPy arrays
X_train = X_train.to_numpy()
X_test = X_test.to_numpy()

# 5. Build FNN model
model = Sequential([
    Input(shape=(784,)),
    Dense(128, activation='relu'),
    Dense(64, activation='relu'),
    Dense(25, activation='softmax')
])

# 6. Compile model
model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

# 7. Train model
model.fit(X_train, y_train, epochs=10, batch_size=32)

# 8. Evaluate model
loss, accuracy = model.evaluate(X_test, y_test)
print("Test Accuracy:", accuracy * 100, "%")

# 9. Predict test images
predictions = model.predict(X_test)

# 10. Display five test images
for i in range(5):

    actual = chr(y_test.iloc[i] + 65)
    predicted = chr(np.argmax(predictions[i]) + 65)

    image = X_test[i].reshape(28, 28)

    plt.imshow(image, cmap='gray')
    plt.title("Actual: " + actual + " | Predicted: " + predicted)
    plt.axis('off')
    plt.show()
